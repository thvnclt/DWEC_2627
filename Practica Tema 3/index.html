function ejercicio3() {
  console.log("--- Ejercicio 3 · Theo Van Celst ---");

  console.log('"10" - 2:', "10" - 2);
  console.log('"10" + 2:', "10" + 2);
  console.log('"4" * 2:', "4" * 2);
  console.log('true + 1:', true + 1);
  console.log('"15" / 3:', "15" / 3);
  console.log('"Item: " + 5:', "Item: " + 5);

  console.log('5 == "5":', 5 == "5");
  console.log('5 === "5":', 5 === "5");
  console.log('0 == false:', 0 == false);
  console.log('0 === false:', 0 === false);
  console.log('null == undefined:', null == undefined);
  console.log('null === undefined:', null === undefined);
}
